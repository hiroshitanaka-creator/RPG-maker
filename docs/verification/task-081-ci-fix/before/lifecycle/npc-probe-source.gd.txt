extends "res://tools/check_region2_village_regression.gd"
func _initialize() -> void:
	contract=WorldExpedition._integers(JSON.parse_string(FileAccess.get_file_as_string("res://docs/verification/task-009/continuing-contract.json")))
	make_base()
	var present := "--npc-present" in OS.get_cmdline_user_args()
	var cell := Vector2i(20,25)
	var adjacent := Vector2i(20,26)
	var saved := placed(0,[4,15])
	var room_data := FirstRegion.room(saved["overworld"])
	check(contract["maps"]["exterior"]["layout"][25][20]==".","NPC正例の固定006セルは床")
	check(room_data["layout"][25][20]==".","配置前後とも本番地形は床")
	check(occupied(saved,cell)==present,"正例NPCの配置有無を明示確認")
	check(FirstRegion.walkable(saved,cell)==not present,"配置前通行可・配置後占有不可")
	check(FirstRegion.walkable(saved,adjacent),"NPC隣接セルは通行可")
	var game := game_for(saved)
	check(walk(game,adjacent),"入口からNPC隣接セルまで本番APIで通常歩行")
	check(game.export_state()["overworld"]["cell"]==[20,26],"NPC隣接への到達を位置で確認")
	var before := game.export_state()
	var moved := game.move_first_region(cell)
	check((moved.is_empty() and game.export_state()==before) if present else (moved.get("kind")=="moved" and game.export_state()["overworld"]["cell"]==[20,25]),"実移動も配置前成功・配置後占有拒否で全状態保持")
	route_log.append({"npc_cell":[20,25],"adjacent_cell":[20,26],"npc_present":present,"terrain":room_data["layout"][25][20],"walkable":FirstRegion.walkable(saved,cell),"adjacent_reached":before["overworld"]["cell"]==[20,26],"move_kind":moved.get("kind",""),"state_unchanged":game.export_state()==before})
	finish()
